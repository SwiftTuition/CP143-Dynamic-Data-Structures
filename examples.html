Example 1: Self-Referential Structures & Basic Nodes

Time: O(1) | Space: O(1)

This example demonstrates how to create self-referential structures and manually link nodes together. Understanding this is fundamental to all dynamic data structures. 

#include <stdio.h>
#include <stdlib.h>

// Define a self-referential structure
typedef struct Node {
    int data;
    struct Node* next;  // Pointer to same type
} Node;

// Function to create a new node
Node* createNode(int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));

    if (newNode == NULL) {
        printf("Memory allocation failed!\n");
        return NULL;
    }

    newNode->data = data;
    newNode->next = NULL;
    return newNode;
}

int main() {
    printf("=== Self-Referential Structures Demo ===\n\n");

    // Create three nodes
    Node* first = createNode(10);
    Node* second = createNode(20);
    Node* third = createNode(30);

    // Link them together manually
    first->next = second;
    second->next = third;
    // third->next is already NULL

    // Traverse and print
    printf("Traversing the linked nodes:\n");
    Node* current = first;
    int position = 1;

    while (current != NULL) {
        printf("Node %d: %d (Address: %p, Next: %p)\n",
               position, current->data, (void*)current, (void*)current->next);
        current = current->next;
        position++;
    }

    // Free memory
    free(first);
    free(second);
    free(third);

    printf("\nMemory freed successfully!\n");
    return 0;
}

Expected Output:

Node 2: 20 (Address: 0x55a8c2c012c0, Next: 0x55a8c2c012e0)
Node 3: 30 (Address: 0x55a8c2c012e0, Next: (nil))

Key Concepts:

  • Self-referential: The struct contains a pointer to its own type
  • malloc(): Allocates memory on the heap, returns pointer
  • NULL checking: Always verify malloc() succeeded
  • Linking: Connect nodes by setting next pointers
  • Traversal: Follow next pointers until reaching NULL

Example 2: Complete Linked List Implementation

Insert: O(1) front, O(n) end | Delete: O(n) | Search: O(n)

A complete linked list implementation with insertion, deletion, searching, and display operations. 

#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

// Insert at beginning - O(1)
void insertAtBeginning(Node** head, int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (newNode == NULL) {
        printf("Memory allocation failed!\n");
        return;
    }

    newNode->data = data;
    newNode->next = *head;  // Point to current head
    *head = newNode;        // Update head
    printf("Inserted %d at beginning\n", data);
}

// Insert at end - O(n)
void insertAtEnd(Node** head, int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (newNode == NULL) {
        printf("Memory allocation failed!\n");
        return;
    }

    newNode->data = data;
    newNode->next = NULL;

    // If list is empty
    if (*head == NULL) {
        *head = newNode;
        printf("Inserted %d (first node)\n", data);
        return;
    }

    // Traverse to end
    Node* current = *head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = newNode;
    printf("Inserted %d at end\n", data);
}

// Delete node with specific value - O(n)
void deleteNode(Node** head, int data) {
    if (*head == NULL) {
        printf("List is empty, cannot delete %d\n", data);
        return;
    }

    // If head needs to be deleted
    if ((*head)->data == data) {
        Node* temp = *head;
        *head = (*head)->next;
        free(temp);
        printf("Deleted %d from beginning\n", data);
        return;
    }

    // Search for node to delete
    Node* current = *head;
    while (current->next != NULL && current->next->data != data) {
        current = current->next;
    }

    if (current->next == NULL) {
        printf("Value %d not found in list\n", data);
        return;
    }

    // Delete the node
    Node* temp = current->next;
    current->next = temp->next;
    free(temp);
    printf("Deleted %d\n", data);
}

// Search for value - O(n)
int search(Node* head, int data) {
    Node* current = head;
    int position = 0;

    while (current != NULL) {
        if (current->data == data) {
            return position;
        }
        current = current->next;
        position++;
    }
    return -1;  // Not found
}

// Display list - O(n)
void displayList(Node* head) {
    if (head == NULL) {
        printf("List: (empty)\n");
        return;
    }

    printf("List: ");
    Node* current = head;
    while (current != NULL) {
        printf("%d", current->data);
        if (current->next != NULL) printf(" -> ");
        current = current->next;
    }
    printf(" -> NULL\n");
}

// Free entire list - O(n)
void freeList(Node** head) {
    Node* current = *head;
    while (current != NULL) {
        Node* temp = current;
        current = current->next;
        free(temp);
    }
    *head = NULL;
    printf("List freed\n");
}

int main() {
    printf("=== Linked List Operations Demo ===\n\n");
    Node* head = NULL;

    // Test insertions
    printf("--- Insertions ---\n");
    insertAtBeginning(&head, 20);
    displayList(head);

    insertAtBeginning(&head, 10);
    displayList(head);

    insertAtEnd(&head, 30);
    displayList(head);

    insertAtEnd(&head, 40);
    displayList(head);

    // Test search
    printf("\n--- Searching ---\n");
    int value = 30;
    int pos = search(head, value);
    if (pos != -1) {
        printf("Found %d at position %d\n", value, pos);
    } else {
        printf("%d not found\n", value);
    }

    // Test deletion
    printf("\n--- Deletions ---\n");
    deleteNode(&head, 20);
    displayList(head);

    deleteNode(&head, 40);
    displayList(head);

    deleteNode(&head, 10);
    displayList(head);

    // Clean up
    printf("\n--- Cleanup ---\n");
    freeList(&head);
    displayList(head);

    return 0;
}

Implementation Notes:

  • Double pointer (Node**): Allows functions to modify the head pointer
  • Insert at beginning: Fast O(1) - just update head
  • Insert at end: Slower O(n) - must traverse entire list
  • Deletion: Handle special case when deleting head node
  • Free entire list: Must free each node individually

Example 3: Stack Implementation (LIFO)

Push: O(1) | Pop: O(1) | Peek: O(1)

A stack using a linked list. All operations are O(1) constant time. 

#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

// Push - O(1)
void push(Node** top, int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (newNode == NULL) {
        printf("Stack overflow! Memory allocation failed.\n");
        return;
    }

    newNode->data = data;
    newNode->next = *top;
    *top = newNode;
    printf("Pushed %d\n", data);
}

// Pop - O(1)
int pop(Node** top) {
    if (*top == NULL) {
        printf("Stack underflow! Stack is empty.\n");
        return -1;
    }

    Node* temp = *top;
    int popped = temp->data;
    *top = (*top)->next;
    free(temp);

    printf("Popped %d\n", popped);
    return popped;
}

// Peek - O(1)
int peek(Node* top) {
    if (top == NULL) {
        printf("Stack is empty.\n");
        return -1;
    }
    return top->data;
}

// Check if empty - O(1)
int isEmpty(Node* top) {
    return top == NULL;
}

// Display stack - O(n)
void displayStack(Node* top) {
    if (top == NULL) {
        printf("Stack: (empty)\n");
        return;
    }

    printf("Stack (top to bottom): ");
    Node* current = top;
    while (current != NULL) {
        printf("%d", current->data);
        if (current->next != NULL) printf(" <- ");
        current = current->next;
    }
    printf("\n");
}

int main() {
    printf("=== Stack Operations Demo (LIFO) ===\n\n");
    Node* stack = NULL;

    // Push operations
    printf("--- Push Operations ---\n");
    push(&stack, 10);
    displayStack(stack);

    push(&stack, 20);
    displayStack(stack);

    push(&stack, 30);
    displayStack(stack);

    push(&stack, 40);
    displayStack(stack);

    // Peek operation
    printf("\n--- Peek Operation ---\n");
    printf("Top element: %d\n", peek(stack));
    displayStack(stack);  // Stack unchanged

    // Pop operations
    printf("\n--- Pop Operations ---\n");
    pop(&stack);
    displayStack(stack);

    pop(&stack);
    displayStack(stack);

    // Check if empty
    printf("\n--- Is Empty? ---\n");
    printf("Stack empty? %s\n", isEmpty(stack) ? "Yes" : "No");

    // Pop remaining
    pop(&stack);
    pop(&stack);
    displayStack(stack);

    printf("Stack empty? %s\n", isEmpty(stack) ? "Yes" : "No");

    // Try popping from empty stack
    printf("\n--- Stack Underflow Test ---\n");
    pop(&stack);

    return 0;
}

Stack Characteristics:

  • LIFO: Last In, First Out - most recently added item is removed first
  • All O(1): Push, pop, and peek all execute in constant time
  • Top pointer: Only need to track the top of the stack
  • Applications: Function call stack, undo/redo, expression evaluation
  • Error handling: Check for overflow (malloc fail) and underflow (empty stack)

Example 4: Queue Implementation (FIFO)

Enqueue: O(1) | Dequeue: O(1) | Front: O(1)

A queue using a linked list with both head and tail pointers for efficient O(1) operations. 

#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct Queue {
    Node* head;  // Front of queue
    Node* tail;  // Rear of queue
} Queue;

// Initialize queue
void initQueue(Queue* q) {
    q->head = NULL;
    q->tail = NULL;
}

// Enqueue - O(1)
void enqueue(Queue* q, int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (newNode == NULL) {
        printf("Queue overflow! Memory allocation failed.\n");
        return;
    }

    newNode->data = data;
    newNode->next = NULL;

    // If queue is empty
    if (q->tail == NULL) {
        q->head = newNode;
        q->tail = newNode;
    } else {
        q->tail->next = newNode;
        q->tail = newNode;
    }

    printf("Enqueued %d\n", data);
}

// Dequeue - O(1)
int dequeue(Queue* q) {
    if (q->head == NULL) {
        printf("Queue underflow! Queue is empty.\n");
        return -1;
    }

    Node* temp = q->head;
    int dequeued = temp->data;
    q->head = q->head->next;

    // If queue becomes empty
    if (q->head == NULL) {
        q->tail = NULL;
    }

    free(temp);
    printf("Dequeued %d\n", dequeued);
    return dequeued;
}

// Get front element - O(1)
int front(Queue* q) {
    if (q->head == NULL) {
        printf("Queue is empty.\n");
        return -1;
    }
    return q->head->data;
}

// Check if empty - O(1)
int isEmpty(Queue* q) {
    return q->head == NULL;
}

// Display queue - O(n)
void displayQueue(Queue* q) {
    if (q->head == NULL) {
        printf("Queue: (empty)\n");
        return;
    }

    printf("Queue (front to rear): ");
    Node* current = q->head;
    while (current != NULL) {
        printf("%d", current->data);
        if (current->next != NULL) printf(" <- ");
        current = current->next;
    }
    printf("\n");
}

int main() {
    printf("=== Queue Operations Demo (FIFO) ===\n\n");
    Queue q;
    initQueue(&q);

    // Enqueue operations
    printf("--- Enqueue Operations ---\n");
    enqueue(&q, 10);
    displayQueue(&q);

    enqueue(&q, 20);
    displayQueue(&q);

    enqueue(&q, 30);
    displayQueue(&q);

    enqueue(&q, 40);
    displayQueue(&q);

    // Front operation
    printf("\n--- Front Operation ---\n");
    printf("Front element: %d\n", front(&q));
    displayQueue(&q);  // Queue unchanged

    // Dequeue operations
    printf("\n--- Dequeue Operations ---\n");
    dequeue(&q);
    displayQueue(&q);

    dequeue(&q);
    displayQueue(&q);

    // Enqueue more
    printf("\n--- Enqueue After Dequeue ---\n");
    enqueue(&q, 50);
    displayQueue(&q);

    enqueue(&q, 60);
    displayQueue(&q);

    // Dequeue all
    printf("\n--- Dequeue All ---\n");
    while (!isEmpty(&q)) {
        dequeue(&q);
        displayQueue(&q);
    }

    // Try dequeuing from empty
    printf("\n--- Queue Underflow Test ---\n");
    dequeue(&q);

    return 0;
}

Queue Characteristics:

  • FIFO: First In, First Out - earliest added item is removed first
  • Two pointers: Head (front) and tail (rear) for O(1) operations
  • Enqueue at tail: Add new elements at the rear
  • Dequeue from head: Remove elements from the front
  • Applications: Task scheduling, breadth-first search, print spooler

Example 5: Binary Search Tree - Complete Implementation

Insert/Search: O(log n) average | Traversal: O(n)

A complete binary search tree with insertion, searching, and three types of traversal. 

#include <stdio.h>
#include <stdlib.h>

typedef struct TreeNode {
    int data;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

// Create new tree node
TreeNode* createNode(int data) {
    TreeNode* newNode = (TreeNode*)malloc(sizeof(TreeNode));
    if (newNode == NULL) {
        printf("Memory allocation failed!\n");
        return NULL;
    }
    newNode->data = data;
    newNode->left = NULL;
    newNode->right = NULL;
    return newNode;
}

// Insert into BST - O(log n) average
TreeNode* insert(TreeNode* root, int data) {
    // Base case: empty tree
    if (root == NULL) {
        return createNode(data);
    }

    // Recursive insertion
    if (data < root->data) {
        root->left = insert(root->left, data);
    } else if (data > root->data) {
        root->right = insert(root->right, data);
    } else {
        printf("Duplicate value %d not inserted\n", data);
    }

    return root;
}

// Search in BST - O(log n) average
TreeNode* search(TreeNode* root, int data) {
    // Base cases
    if (root == NULL || root->data == data) {
        return root;
    }

    // Recursive search
    if (data < root->data) {
        return search(root->left, data);
    } else {
        return search(root->right, data);
    }
}

// Inorder traversal: Left -> Root -> Right (sorted order)
void inorder(TreeNode* root) {
    if (root != NULL) {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}

// Preorder traversal: Root -> Left -> Right
void preorder(TreeNode* root) {
    if (root != NULL) {
        printf("%d ", root->data);
        preorder(root->left);
        preorder(root->right);
    }
}

// Postorder traversal: Left -> Right -> Root
void postorder(TreeNode* root) {
    if (root != NULL) {
        postorder(root->left);
        postorder(root->right);
        printf("%d ", root->data);
    }
}

// Find minimum value
TreeNode* findMin(TreeNode* root) {
    if (root == NULL) return NULL;
    while (root->left != NULL) {
        root = root->left;
    }
    return root;
}

// Find maximum value
TreeNode* findMax(TreeNode* root) {
    if (root == NULL) return NULL;
    while (root->right != NULL) {
        root = root->right;
    }
    return root;
}

// Count nodes
int countNodes(TreeNode* root) {
    if (root == NULL) return 0;
    return 1 + countNodes(root->left) + countNodes(root->right);
}

// Calculate height
int height(TreeNode* root) {
    if (root == NULL) return -1;

    int leftHeight = height(root->left);
    int rightHeight = height(root->right);

    return 1 + (leftHeight > rightHeight ? leftHeight : rightHeight);
}

// Free tree (postorder)
void freeTree(TreeNode* root) {
    if (root != NULL) {
        freeTree(root->left);
        freeTree(root->right);
        free(root);
    }
}

int main() {
    printf("=== Binary Search Tree Demo ===\n\n");
    TreeNode* root = NULL;

    // Insert nodes
    printf("--- Inserting Nodes ---\n");
    int values[] = {50, 30, 70, 20, 40, 60, 80, 10, 25, 35, 65};
    int n = sizeof(values) / sizeof(values[0]);

    for (int i = 0; i < n; i++) {
        printf("Inserting %d\n", values[i]);
        root = insert(root, values[i]);
    }

    // Tree structure
    printf("\nTree structure (root is 50):\n");
    printf("              50\n");
    printf("            /    \\\n");
    printf("          30      70\n");
    printf("         /  \\    /  \\\n");
    printf("       20   40  60   80\n");
    printf("      / \\   /    \\\n");
    printf("    10  25 35    65\n\n");

    // Traversals
    printf("--- Traversals ---\n");
    printf("Inorder (sorted):   ");
    inorder(root);
    printf("\n");

    printf("Preorder:           ");
    preorder(root);
    printf("\n");

    printf("Postorder:          ");
    postorder(root);
    printf("\n\n");

    // Search operations
    printf("--- Search Operations ---\n");
    int searchValues[] = {40, 55, 10};
    for (int i = 0; i < 3; i++) {
        TreeNode* result = search(root, searchValues[i]);
        if (result != NULL) {
            printf("Found %d in tree\n", searchValues[i]);
        } else {
            printf("%d not found in tree\n", searchValues[i]);
        }
    }

    // Tree statistics
    printf("\n--- Tree Statistics ---\n");
    TreeNode* minNode = findMin(root);
    TreeNode* maxNode = findMax(root);
    printf("Minimum value: %d\n", minNode ? minNode->data : -1);
    printf("Maximum value: %d\n", maxNode ? maxNode->data : -1);
    printf("Total nodes: %d\n", countNodes(root));
    printf("Tree height: %d\n", height(root));

    // Cleanup
    printf("\n--- Cleanup ---\n");
    freeTree(root);
    printf("Tree freed\n");

    return 0;
}

BST Properties & Traversals:

  • BST Property: Left subtree < Node < Right subtree (always maintained)
  • Inorder: Visits nodes in sorted ascending order
  • Preorder: Useful for copying tree structure
  • Postorder: Used for freeing tree (delete children before parent)
  • Balance matters: Unbalanced trees degrade to O(n) performance
  • Recursion: Tree operations naturally expressed recursively
